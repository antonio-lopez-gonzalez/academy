Cheatsheet de comandos Redis

Referencia rápida 📋 Redis 7+ · StackExchange.Redis

Keys y expiración

SET user:42:name ana             # string con nombre
SET hub:conn:8632 node-1 EX 30   # + expira en 30 s (presencia)
EXPIRE hub:conn:8632 30          # fija/refresca el TTL
TTL hub:conn:8632                # segundos restantes (-1 = no expira)
PERSIST hub:conn:8632            # quita el TTL
DEL hub:conn:8632                # borra la key
EXISTS hub:conn:8632             # 1 si existe
KEYS hub:conn:*                  # ⚠ lista todas (solo dev, usa SCAN)
SCAN 0 MATCH hub:conn:* COUNT 100

Strings (caché)

SET cached:price:msft 415.25            # insertar
GET cached:price:msft                    # leer
SET cached:user:1 '{"id":1}' EX 60       # JSON con TTL
SETNX token:user:1 abc123                # solo si NO existe (NX)
SET cached:price:msft 420 NX             # SIN sobrescribir si existe
MGET cached:price:msft cached:price:aapl # leer varios
INCR counter:msgs                        # 1, 2, 3… (atómico)
INCRBY counter:msgs 5
DECR counter:msgs
APPEND log:line "..."

Hashes (un objeto)

HSET hub:conn:8632 node node-1 state open userId 42
HGET hub:conn:8632 node
HGETALL hub:conn:8632     # todas las parejas campo:valor
HEXISTS hub:conn:8632 node
HDEL hub:conn:8632 state
HLEN hub:conn:8632
HINCRBY hub:conn:8632 msgs 1   # campo contador

Lists (colas)

LPUSH queue:trades '{"id":887}'    # mete por la izquierda
RPOP queue:trades                   # saca por la derecha (FIFO)
LPUSH job:mail ... | BRPOP job:mail 0   # bloqueante (espera sin loop)
LRANGE queue:trades 0 -1            # todos
LLEN queue:trades                   # tamaño
LTRIM queue:trades 0 99             # recorta (búfer acotado)

Sets (miembros únicos)

SADD chan:gaming user:42 user:7
SMEMBERS chan:gaming
SISMEMBER chan:gaming user:42       # ¿está?
SCARD chan:gaming                   # nº de miembros
SREM chan:gaming user:7
SUNION chan:gaming chan:news

Sorted sets (ranking / ventana temporal)

ZADD presence 1750000000 user:42          # score = timestamp
ZSCORE presence user:42
ZRANGEBYSCORE presence 1749913600 +inf    # conectados desde hace X
ZREM presence user:42                     # desconexión
ZRANGEBYSCORE presence -inf +inf WITHSCORES
# rate limiting por ventana:
SETEX rl:user:42:1728000000 60 1
INCR rl:user:42:1728000000
EXPIRE rl:user:42:1728000000 60

Pub/Sub (broadcast entre nodos)

# Terminal A (suscriptor)
SUBSCRIBE hub:events
# Terminal B (publica)
PUBLISH hub:events '{"type":"trade","node":"node-1"}'
# patrón:
PSUBSCRIBE hub:*
# ver canales/oyentes:
PUBSUB CHANNELS
PUBSUB NUMSUB hub:events

Streams (cola robusta)

XADD trades * type trade id 887      # añade (id auto)
XLEN trades
XRANGE trades - + COUNT 10           # leer primeros
XREAD COUNT 5 BLOCK 0 STREAMS trades $   # nuevo desde ahora (bloqueante)
XGROUP CREATE trades workers 0 MKSTREAM   # grupo de consumidores
XREADGROUP GROUP workers Alice COUNT 10 STREAMS trades >
XACK trades workers ID              # confirmar procesado
XTRIM trades MAXLEN ~ 1000          # acotar tamaño

Administración

PING
INFO                       # stats, memoria, replicación
INFO replication
CONFIG GET maxmemory
CONFIG SET maxmemory-policy allkeys-lru   # eviction
DBSIZE
FLUSHDB                    # ⚠ borra la DB actual (dev)
SELECT 3                   # cambiar DB lógica (0-15)
AUTH <password>
CLIENT LIST
BGSAVE                     # snapshot RDB
SLAVEOF host port          # réplica (config)
REPLICAOF host port        # réplica (7.x)

StackExchange.Redis (lo esencial)

using StackExchange.Redis;

// 1. Un solo ConnectionMultiplexer por proceso (DI singleton)
var mux = ConnectionMultiplexer.Connect("localhost:6379");
IDatabase db = mux.GetDatabase();

// 2. Strings
await db.StringSetAsync("hub:conn:8632", "node-1", TimeSpan.FromSeconds(30));
string? node = await db.StringGetAsync("hub:conn:8632");
long count = await db.StringIncrementAsync("counter:msgs");

// 3. Hashes
await db.HashSetAsync("hub:conn:8632",
    [new HashEntry("node", "node-1"), new HashEntry("userId", "42")]);
var conn = await db.HashGetAllAsync("hub:conn:8632");

// 4. Sets
await db.SetAddAsync("chan:gaming", "user:42");
bool online = await db.SetContainsAsync("chan:gaming", "user:42");

// 5. Sorted sets (presencia con timestamp)
await db.SortedSetAddAsync("presence", "user:42", DateTimeOffset.UtcNow.ToUnixTimeSeconds());
var active = await db.SortedSetRangeByScoreAsync("presence", cutoff, double.PositiveInfinity);

// 6. Pub/Sub
ISubscriber sub = mux.GetSubscriber();
await sub.SubscribeAsync("hub:events", (channel, value) => Handle(value!));
await sub.PublishAsync("hub:events", "{\"type\":\"trade\"}");

// 7. Expiración
await db.KeyExpireAsync("hub:conn:8632", TimeSpan.FromSeconds(30));
// 8. Caché serializada (IDistributedCache)
builder.Services.AddStackExchangeRedisCache(o => o.Configuration = "localhost:6379");

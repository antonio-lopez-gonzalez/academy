3 · Paso 1: Capa de servicios

src/services/api.js
const API_BASE = "https://rickandmortyapi.com/api";

export async function apiGet(endpoint, params = {}) {
  const searchParams = new URLSearchParams(params);
  const url = `${API_BASE}${endpoint}?${searchParams}`;

  const response = await fetch(url);
  if (!response.ok) {
    throw new Error(`Error ${response.status}: ${response.statusText}`);
  }
  return response.json();
}
src/services/characterService.js
import { apiGet } from "./api";

export const characterService = {
  getAll(params = {}) {
    return apiGet("/character", params);
  },

  getById(id) {
    return apiGet(`/character/${id}`);
  },

  search(query) {
    return apiGet("/character", { name: query });
  },
};

src/components/FilterPanel.jsx
export default function FilterPanel({ filters, onFilterChange }) {
  return (
    <div className="filter-panel">
      <div>
        <label>Estado:</label>
        <select
          value={filters.status}
          onChange={(e) => onFilterChange({ ...filters, status: e.target.value })}
        >
          <option value="">Todos</option>
          <option value="alive">Vivo</option>
          <option value="dead">Muerto</option>
          <option value="unknown">Desconocido</option>
        </select>
      </div>

      <div>
        <label>Género:</label>
        <select
          value={filters.gender}
          onChange={(e) => onFilterChange({ ...filters, gender: e.target.value })}
        >
          <option value="">Todos</option>
          <option value="female">Femenino</option>
          <option value="male">Masculino</option>
          <option value="genderless">Sin género</option>
          <option value="unknown">Desconocido</option>
        </select>
      </div>
    </div>
  );
}
src/components/Pagination.jsx
export default function Pagination({ info, currentPage, onPageChange }) {
  if (!info || info.pages <= 1) return null;

  return (
    <div className="pagination">
      <button
        onClick={() => onPageChange(currentPage - 1)}
        disabled={currentPage === 1}
      >
        Anterior
      </button>
      <span>
        Página {currentPage} de {info.pages}
      </span>
      <button
        onClick={() => onPageChange(currentPage + 1)}
        disabled={currentPage === info.pages}
      >
        Siguiente
      </button>
    </div>
  );
}

5 · Paso 3: Páginas

src/pages/Home.jsx
import { useState, useEffect } from "react";
import { characterService } from "../services/characterService";
import CharacterCard from "../components/CharacterCard";
import FilterPanel from "../components/FilterPanel";
import Pagination from "../components/Pagination";

export default function Home() {
  const [characters, setCharacters] = useState([]);
  const [info, setInfo] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);
  const [currentPage, setCurrentPage] = useState(1);
  const [filters, setFilters] = useState({ status: "", gender: "" });
  const [search, setSearch] = useState("");

  useEffect(() => {
    const controller = new AbortController();

    async function loadCharacters() {
      setLoading(true);
      setError(null);
      try {
        const params = { page: currentPage };
        if (filters.status) params.status = filters.status;
        if (filters.gender) params.gender = filters.gender;
        if (search) params.name = search;

        const data = await characterService.getAll(params);
        setCharacters(data.results);
        setInfo(data.info);
      } catch (err) {
        if (err.name !== "AbortError") {
          setError(err.message);
        }
      } finally {
        setLoading(false);
      }
    }

    loadCharacters();

    return () => controller.abort();
  }, [currentPage, filters, search]);

  return (
    <div>
      <h1>Rick and Morty Explorer</h1>

      <input
        type="search"
        placeholder="Buscar personaje..."
        value={search}
        onChange={(e) => {
          setSearch(e.target.value);
          setCurrentPage(1);
        }}
      />

      <FilterPanel filters={filters} onFilterChange={setFilters} />

      {loading && <p>Cargando personajes...</p>}
      {error && <p style={{ color: "red" }}>Error: {error}</p>}

      {!loading && !error && (
        <>
          <div className="character-grid">
            {characters.map((character) => (
              <CharacterCard key={character.id} character={character} />
            ))}
          </div>
          <Pagination
            info={info}
            currentPage={currentPage}
            onPageChange={setCurrentPage}
          />
        </>
      )}
    </div>
  );
}

10 · Preguntas frecuentes

¿Puedo usar otra API pública?
Sí: PokéAPI, The Movie DB, OpenWeatherMap, REST Countries. Todas gratuitas y sin key (o con key gratuita).
¿Cómo manejo rate limiting?
Algunas APIs limitan peticiones por segundo. Añade delays entre peticiones o usa caché.
¿Puedo cachear respuestas?
Sí, usa localStorage o una librería como React Query para cachear respuestas de API.
¿Cómo manejo APIs que requieren autenticación?
Regístrate para obtener una API key y envíala en los headers o query params según la documentación de la API.

fetch y async/await en React

Módulo 9 · Consumo de APIs ⏱ 40 minutos

1 · Introducción

Las aplicaciones reales necesitan comunicarse con servidores: cargar datos, enviar formularios, autenticar usuarios. La API nativa de JavaScript para hacer peticiones HTTP es fetch. Combinada con async/await, hace que el código asíncrono sea legible y fácil de manejar.

En React, las peticiones a APIs se hacen dentro de useEffect (para cargar datos al montar el componente) o en manejadores de eventos (para enviar datos). En esta lección veremos cómo usar fetch correctamente en componentes React.

2 · Teoría

Sintaxis básica de fetch

// GET request
fetch("https://api.example.com/data")
  .then(response => {
    if (!response.ok) {
      throw new Error(`HTTP error! status: ${response.status}`);
    }
    return response.json();
  })
  .then(data => console.log(data))
  .catch(error => console.error("Error:", error));

Versión con async/await

async function fetchData() {
  try {
    const response = await fetch("https://api.example.com/data");
    if (!response.ok) {
      throw new Error(`HTTP error! status: ${response.status}`);
    }
    const data = await response.json();
    console.log(data);
  } catch (error) {
    console.error("Error:", error);
  }
}

fetch en useEffect

import { useState, useEffect } from "react";

function UserList() {
  const [users, setUsers] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    async function loadUsers() {
      try {
        setLoading(true);
        const response = await fetch("https://jsonplaceholder.typicode.com/users");
        if (!response.ok) throw new Error("Error al cargar usuarios");
        const data = await response.json();
        setUsers(data);
        setError(null);
      } catch (err) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    }

    loadUsers();
  }, []);

  if (loading) return <p>Cargando...</p>;
  if (error) return <p>Error: {error}</p>;

  return (
    <ul>
      {users.map(user => (
        <li key={user.id}>{user.name}</li>
      ))}
    </ul>
  );
}

Métodos HTTP

// POST
await fetch("/api/users", {
  method: "POST",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ name: "Ana", email: "ana@example.com" }),
});

// PUT (actualizar completo)
await fetch(`/api/users/${id}`, {
  method: "PUT",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ name: "Ana", email: "ana.nueva@example.com" }),
});

// PATCH (actualizar parcial)
await fetch(`/api/users/${id}`, {
  method: "PATCH",
  headers: { "Content-Type": "application/json" },
  body: JSON.stringify({ email: "ana.nueva@example.com" }),
});

// DELETE
await fetch(`/api/users/${id}`, { method: "DELETE" });

4 · Ejemplo real

Formulario que envía datos a una API:

src/components/CreateUser.jsx
import { useState } from "react";

export default function CreateUser() {
  const [name, setName] = useState("");
  const [email, setEmail] = useState("");
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState(null);
  const [success, setSuccess] = useState(false);

  async function handleSubmit(e) {
    e.preventDefault();
    setLoading(true);
    setError(null);
    setSuccess(false);

    try {
      const response = await fetch("https://jsonplaceholder.typicode.com/users", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ name, email }),
      });

      if (!response.ok) throw new Error("Error al crear usuario");

      const data = await response.json();
      console.log("Usuario creado:", data);
      setSuccess(true);
      setName("");
      setEmail("");
    } catch (err) {
      setError(err.message);
    } finally {
      setLoading(false);
    }
  }

  return (
    <form onSubmit={handleSubmit}>
      <h2>Crear usuario</h2>
      {error && <p style={{ color: "red" }}>{error}</p>}
      {success && <p style={{ color: "green" }}>¡Usuario creado!</p>}

      <div>
        <label>Nombre:</label>
        <input
          value={name}
          onChange={(e) => setName(e.target.value)}
          required
        />
      </div>

      <div>
        <label>Email:</label>
        <input
          type="email"
          value={email}
          onChange={(e) => setEmail(e.target.value)}
          required
        />
      </div>

      <button type="submit" disabled={loading}>
        {loading ? "Creando..." : "Crear usuario"}
      </button>
    </form>
  );
}

6 · Errores habituales

Error 1: no verificar response.ok

const response = await fetch("/api/data");
const data = await response.json();   // ✘ si response.status es 404, json() puede fallar

const response = await fetch("/api/data");
if (!response.ok) throw new Error("Error");   // ✔ verificar antes
const data = await response.json();

Error 2: olvidar el estado de carga

function DataComponent() {
  const [data, setData] = useState(null);

  useEffect(() => {
    fetch("/api/data").then(r => r.json()).then(setData);
  }, []);

  return <div>{data.map(...)}</div>;   // ✘ error si data es null
}

Error 3: no manejar errores

async function loadData() {
  const response = await fetch("/api/data");
  const data = await response.json();   // ✘ si falla, la promesa se rechaza sin catch
}

9 · Resumen

  • fetch es la API nativa para hacer peticiones HTTP.
  • Usa async/await para código más legible.
  • Siempre verifica response.ok antes de procesar la respuesta.
  • Maneja errores con try/catch.
  • Muestra estados de carga y error en la UI.
  • Usa useEffect para cargar datos al montar el componente.

10 · Preguntas frecuentes

¿Puedo usar axios en lugar de fetch?
Sí, axios es una librería popular que simplifica las peticiones. Pero fetch es nativo y suficiente para la mayoría de casos.
¿Cómo envío archivos (multipart/form-data)?
Usa FormData: const formData = new FormData(); formData.append("file", file); y pásalo como body.
¿Cómo envío headers de autenticación?
headers: { "Authorization": `Bearer ${token}` }.
¿Puedo hacer peticiones en paralelo?
Sí, con Promise.all(): const [users, posts] = await Promise.all([fetchUsers(), fetchPosts()]);.

useState

import { useState } from 'react';

export function Counter() {
  const [count, setCount] = useState(0);

  return (
    <button onClick={() => setCount(c => c + 1)}>
      Contador: {count}
    </button>
  );
}
  • Nunca modifiques el estado directamente: usa la función setter.
  • Si el nuevo estado depende del anterior, pasa una función.

useEffect

import { useEffect, useState } from 'react';

export function User({ id }) {
  const [user, setUser] = useState(null);

  useEffect(() => {
    const controller = new AbortController();
    fetch(`/api/users/${id}`, { signal: controller.signal })
      .then(r => r.json())
      .then(setUser);

    return () => controller.abort();
  }, [id]);

  return <p>{user?.name ?? 'Cargando...'}</p>;
}

useMemo y useCallback

import { useMemo, useCallback } from 'react';

export function List({ items, filter }) {
  const filtered = useMemo(
    () => items.filter(i => i.includes(filter)),
    [items, filter]
  );

  const handleClick = useCallback((id) => {
    console.log('Click en', id);
  }, []);

  return (
    <ul>
      {filtered.map(item => (
        <li key={item.id} onClick={() => handleClick(item.id)}>
          {item.name}
        </li>
      ))}
    </ul>
  );
}

useReducer

import { useReducer } from 'react';

const initial = { count: 0 };

function reducer(state, action) {
  switch (action.type) {
    case 'increment': return { count: state.count + 1 };
    case 'decrement': return { count: state.count - 1 };
    default: return state;
  }
}

export function Counter() {
  const [state, dispatch] = useReducer(reducer, initial);

  return (
    <>
      <p>{state.count}</p>
      <button onClick={() => dispatch({ type: 'increment' })}>+</button>
      <button onClick={() => dispatch({ type: 'decrement' })}>-</button>
    </>
  );
}

Context API

// src/context/ThemeContext.jsx
import { createContext, useContext, useState } from 'react';

const ThemeContext = createContext(null);

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState('light');
  return (
    <ThemeContext.Provider value={{ theme, setTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme() {
  const ctx = useContext(ThemeContext);
  if (!ctx) throw new Error('useTheme debe usarse dentro de ThemeProvider');
  return ctx;
}

Custom hook

import { useEffect, useState } from 'react';

export function useFetch(url) {
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    const controller = new AbortController();
    setLoading(true);
    fetch(url, { signal: controller.signal })
      .then(r => { if (!r.ok) throw new Error(r.status); return r.json(); })
      .then(setData)
      .catch(setError)
      .finally(() => setLoading(false));
    return () => controller.abort();
  }, [url]);

  return { data, loading, error };
}

React 19: useActionState

import { useActionState } from "react";

async function save(previousState, formData) {
  if (!formData.get("name")) {
    return { ...previousState, error: "Falta el nombre" };
  }
  await api.save(formData);
  return { error: "", success: true };
}

export function NameForm() {
  const [state, formAction, isPending] = useActionState(save, { error: "", success: false });

  return (
    <form action={formAction}>
      <input name="name" />
      {state.error && <p role="alert">{state.error}</p>}
      <button disabled={isPending}>{isPending ? "Guardando…" : "Guardar"}</button>
    </form>
  );
}
  • Devuelve [state, formAction, isPending].
  • La action recibe (previousState, formData) y devuelve el nuevo estado.
  • Sustituye al antiguo useFormState (nombre canary). Importa desde "react".

React 19: useDeferredValue

import { useDeferredValue, useMemo, useState } from "react";

function Search({ items }) {
  const [query, setQuery] = useState("");
  const deferredQuery = useDeferredValue(query, ""); // initialValue (React 19)

  const filtered = useMemo(
    () => items.filter((i) => i.toLowerCase().includes(deferredQuery.toLowerCase())),
    [deferredQuery, items]
  );

  return (
    <>
      <input value={query} onChange={(e) => setQuery(e.target.value)} />
      <ul>{filtered.map((i) => <li key={i}>{i}</li>)}</ul>
    </>
  );
}
  • El input responde al instante; el filtrado se difiere.
  • Úsalo cuando el valor viene "de fuera" (prop/estado del padre). Si controlas la actualización, prefiere useTransition.

React 19: useId

import { useId } from "react";

export function Field({ label, ...props }) {
  const id = useId();

    <>

React 19.2: useEffectEvent

import { useEffect, useEffectEvent } from "react";

function Chat({ roomId, theme }) {
  const onConnected = useEffectEvent(() => {
    showNotification("Conectado", theme); // ve el theme actual
  });

  useEffect(() => {
    const conn = createConnection(roomId);
    conn.on("connected", onConnected);
    conn.connect();
    return () => conn.disconnect();
  }, [roomId]); // theme NO va en las dependencias

  return <div />;
}
  • Funciones que siempre ven los últimos props/estado sin reiniciar el efecto.
  • Solo se llaman desde efectos. No van en las dependencias.

React 19.2: Activity

import { Activity, useState } from "react";

export function Tabs() {
  const [tab, setTab] = useState("a");

React 19: Document Metadata y preloading

import { preload, preinit, preconnect } from "react-dom";

export function ProductPage() {
  preconnect("https://api.ejemplo.com");
  preload("/fonts/Inter.woff2", { as: "font" });
  preinit("/styles/product.css", { as: "style" });

  return (
    <>
      <title>Producto · Mi Tienda</title>
      <meta name="description" content="Descripción del producto" />
      <h1>Producto</h1>
    </>
  );
}
  • React "sube" <title>, <meta>, <link> y <script> al <head>.
  • preload descarga sin ejecutar; preinit descarga y ejecuta.
  • preconnect abre conexión temprana; prefetchDNS resuelve el DNS.

Reglas de los hooks

  1. Solo en componentes de React o custom hooks.
  2. Siempre en el nivel superior, nunca dentro de condicionales, bucles o funciones anidadas.
  3. Siempre en el mismo orden en cada render.

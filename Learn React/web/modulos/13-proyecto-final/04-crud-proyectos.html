CRUD de Proyectos

Módulo 13 · Proyecto Final

Introducción

Completaremos el CRUD de proyectos con las operaciones de actualizar y eliminar, añadiendo confirmaciones y manejo de errores.

Detalle de proyecto

// src/pages/ProjectDetail.jsx
import { useParams, useNavigate } from 'react-router-dom'
import { useProject } from '../hooks/useProject'
import TaskBoard from '../components/tasks/TaskBoard'
import styles from './ProjectDetail.module.css'

export default function ProjectDetail() {
  const { id } = useParams()
  const navigate = useNavigate()
  const { project, loading, error, deleteProject } = useProject(id)

  async function handleDelete() {
    if (window.confirm('¿Eliminar este proyecto?')) {
      await deleteProject()
      navigate('/')
    }
  }

  if (loading) return <div>Cargando...</div>
  if (error) return <div>{error}</div>
  if (!project) return <div>Proyecto no encontrado</div>

  return (
    <div className={styles.detail}>
      <div className={styles.header}>
        <h1>{project.name}</h1>
        <p>{project.description}</p>
        <button onClick={handleDelete} className={styles.deleteBtn}>
          Eliminar Proyecto
        </button>
      </div>
      
      <TaskBoard projectId={id} />
    </div>
  )
}

Hook useProject

// src/hooks/useProject.js
import { useState, useEffect, useCallback } from 'react'
import { projectService } from '../services/projectService'

export function useProject(id) {
  const [project, setProject] = useState(null)
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  async function loadProject() {
    try {
      setLoading(true)
      const data = await projectService.getById(id)
      setProject(data)
      setError(null)
    } catch (err) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    loadProject()
  }, [id])

  const updateProject = useCallback(async (updates) => {
    const updated = await projectService.update(id, updates)
    setProject(updated)
    return updated
  }, [id])

  const deleteProject = useCallback(async () => {
    await projectService.delete(id)
  }, [id])

  return {
    project,
    loading,
    error,
    updateProject,
    deleteProject,
    reload: loadProject
  }
}

Editar proyecto

// src/components/projects/EditProjectModal.jsx
import { useState, useEffect } from 'react'
import styles from './EditProjectModal.module.css'

export default function EditProjectModal({ project, onClose, onSubmit }) {
  const [name, setName] = useState('')
  const [description, setDescription] = useState('')
  const [loading, setLoading] = useState(false)

  useEffect(() => {
    if (project) {
      setName(project.name)
      setDescription(project.description || '')
    }
  }, [project])

  async function handleSubmit(e) {
    e.preventDefault()
    setLoading(true)
    
    try {
      await onSubmit({ name, description })
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className={styles.overlay} onClick={onClose}>
      <div className={styles.modal} onClick={e => e.stopPropagation()}>
        <h2>Editar Proyecto</h2>
        <form onSubmit={handleSubmit}>
          <div className={styles.field}>
            <label>Nombre</label>
            <input
              type="text"
              value={name}
              onChange={(e) => setName(e.target.value)}
              required
            />
          </div>

          <div className={styles.field}>
            <label>Descripción</label>
            <textarea
              value={description}
              onChange={(e) => setDescription(e.target.value)}
              rows={3}
            />
          </div>

          <div className={styles.actions}>
            <button type="button" onClick={onClose}>
              Cancelar
            </button>
            <button type="submit" disabled={loading}>
              {loading ? 'Guardando...' : 'Guardar'}
            </button>
          </div>
        </form>
      </div>
    </div>
  )
}

Comentarios y Subtareas

Módulo 13 · Proyecto Final

Introducción

Una tarea profesional no vive sola. Añadiremos comentarios para que el equipo comunique avances y subtareas para dividir el trabajo en pasos pequeños y medibles.

Servicios adicionales

// src/services/commentService.js
import { api } from './api'

export const commentService = {
  getByTask: (taskId) =>
    api.get(`/tasks/${taskId}/comments`),

  create: (taskId, content) =>
    api.post(`/tasks/${taskId}/comments`, { content }),

  delete: (id) =>
    api.delete(`/comments/${id}`)
}

// src/services/subtaskService.js
import { api } from './api'

export const subtaskService = {
  getByTask: (taskId) =>
    api.get(`/tasks/${taskId}/subtasks`),

  create: (taskId, content) =>
    api.post(`/tasks/${taskId}/subtasks`, { content }),

  toggle: (id, done) =>
    api.put(`/subtasks/${id}`, { done }),

  delete: (id) =>
    api.delete(`/subtasks/${id}`)
}

Hook useTaskDetail

// src/hooks/useTaskDetail.js
import { useState, useEffect, useCallback } from 'react'
import { taskService } from '../services/taskService'
import { commentService } from '../services/commentService'
import { subtaskService } from '../services/subtaskService'

export function useTaskDetail(taskId) {
  const [task, setTask] = useState(null)
  const [comments, setComments] = useState([])
  const [subtasks, setSubtasks] = useState([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  async function load() {
    try {
      setLoading(true)
      const [taskData, commentsData, subtasksData] = await Promise.all([
        taskService.getById(taskId),
        commentService.getByTask(taskId),
        subtaskService.getByTask(taskId)
      ])
      setTask(taskData)
      setComments(commentsData)
      setSubtasks(subtasksData)
      setError(null)
    } catch (err) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    load()
  }, [taskId])

  const addComment = useCallback(async (content) => {
    const comment = await commentService.create(taskId, content)
    setComments((prev) => [...prev, comment])
  }, [taskId])

  const addSubtask = useCallback(async (content) => {
    const subtask = await subtaskService.create(taskId, content)
    setSubtasks((prev) => [...prev, subtask])
  }, [taskId])

  const toggleSubtask = useCallback(async (id, done) => {
    const updated = await subtaskService.toggle(id, done)
    setSubtasks((prev) => prev.map((s) => (s.id === id ? updated : s)))
  }, [])

  const deleteSubtask = useCallback(async (id) => {
    await subtaskService.delete(id)
    setSubtasks((prev) => prev.filter((s) => s.id !== id))
  }, [])

  const completedCount = subtasks.filter((s) => s.done).length
  const progress = subtasks.length > 0
    ? Math.round((completedCount / subtasks.length) * 100)
    : 0

  return {
    task,
    comments,
    subtasks,
    loading,
    error,
    progress,
    addComment,
    addSubtask,
    toggleSubtask,
    deleteSubtask,
    reload: load
  }
}

Página de detalle de tarea

// src/pages/TaskDetail.jsx
import { useParams, useNavigate } from 'react-router-dom'
import { useTaskDetail } from '../hooks/useTaskDetail'
import CommentList from '../components/tasks/CommentList'
import SubtaskList from '../components/tasks/SubtaskList'
import styles from './TaskDetail.module.css'

export default function TaskDetail() {
  const { id } = useParams()
  const navigate = useNavigate()
  const {
    task,
    comments,
    subtasks,
    loading,
    progress,
    addComment,
    addSubtask,
    toggleSubtask,
    deleteSubtask
  } = useTaskDetail(id)

  if (loading) return <div>Cargando...</div>
  if (!task) return <div>Tarea no encontrada</div>

  return (
    <div className={styles.detail}>
      <button onClick={() => navigate(-1)}>← Volver</button>

      <h1>{task.title}</h1>
      <p>{task.description}</p>

      <div className={styles.progress}>
        <label>Progreso: {progress}%</label>
        <progress value={progress} max={100} />
      </div>

      <SubtaskList
        subtasks={subtasks}
        onAdd={addSubtask}
        onToggle={toggleSubtask}
        onDelete={deleteSubtask}
      />

      <CommentList
        comments={comments}
        onAdd={addComment}
      />
    </div>
  )
}

Lista de subtareas

// src/components/tasks/SubtaskList.jsx
import { useState } from 'react'
import styles from './SubtaskList.module.css'

export default function SubtaskList({ subtasks, onAdd, onToggle, onDelete }) {
  const [content, setContent] = useState('')

  async function handleSubmit(e) {
    e.preventDefault()
    if (!content.trim()) return
    await onAdd(content)
    setContent('')
  }

  return (
    <section className={styles.section}>
      <h3>Subtareas</h3>

      <form onSubmit={handleSubmit} className={styles.form}>
        <input
          value={content}
          onChange={(e) => setContent(e.target.value)}
          placeholder="Añadir subtarea"
        />
        <button type="submit">Añadir</button>
      </form>

      <ul className={styles.list}>
        {subtasks.map((subtask) => (
          <li key={subtask.id} className={subtask.done ? styles.done : ''}>
            <label>
              <input
                type="checkbox"
                checked={subtask.done}
                onChange={(e) => onToggle(subtask.id, e.target.checked)}
              />
              {subtask.content}
            </label>
            <button onClick={() => onDelete(subtask.id)}>Eliminar</button>
          
        ))}
      
    </section>
  )
}

Lista de comentarios

// src/components/tasks/CommentList.jsx
import { useState } from 'react'
import styles from './CommentList.module.css'

export default function CommentList({ comments, onAdd }) {
  const [content, setContent] = useState('')

  async function handleSubmit(e) {
    e.preventDefault()
    if (!content.trim()) return
    await onAdd(content)
    setContent('')
  }

  return (
    <section className={styles.section}>
      <h3>Comentarios</h3>

      <form onSubmit={handleSubmit} className={styles.form}>
        <textarea
          value={content}
          onChange={(e) => setContent(e.target.value)}
          placeholder="Escribe un comentario"
          rows={3}
        />
        <button type="submit">Comentar</button>
      </form>

      <ul className={styles.list}>
        {comments.map((comment) => (
          <li key={comment.id}>
            <div className={styles.meta}>
              <strong>{comment.authorName}</strong>
              <time>{new Date(comment.createdAt).toLocaleString()}</time>
            </div>
            <p>{comment.content}</p>
          
        ))}
      
    </section>
  )
}

Buenas prácticas

  • Carga paralela con Promise.all para reducir esperas
  • La barra de progreso deriva su valor del estado, no necesita estado propio
  • Cada lista maneja su propio input local; solo emite el valor final
  • Evita anidar arrays profundos en el estado global; usa hooks locales

Resumen

  • useTaskDetail agrupa tarea, comentarios y subtareas en un solo hook
  • Promise.all carga los tres recursos en paralelo
  • SubtaskList permite añadir, marcar y eliminar subtareas
  • CommentList muestra conversaciones ordenadas por fecha
  • El progreso se calcula a partir del porcentaje de subtareas completadas

Sidebar

// src/components/layout/Sidebar.jsx
import { Link, useLocation } from 'react-router-dom'
import { useProjects } from '../../hooks/useProjects'

export default function Sidebar() {
  const location = useLocation()

  return (
    <aside className={styles.sidebar}>
      <nav>
        <Link 
          to="/" 
          className={location.pathname === '/' ? styles.active : ''}
        >
          Dashboard
        </Link>
        
        <div className={styles.section}>
          <h3>Proyectos</h3>
          <ul>
            {projects?.map(project => (

                  to={`/projects/${project.id}`}
                  className={location.pathname.includes(project.id) ? styles.active : ''}
                >
                  {project.name}
                </Link>
              </li>
            ))}
          </ul>
        </div>
      </nav>
    </aside>
  )
}

Página Dashboard

// src/pages/Dashboard.jsx
import { useState } from 'react'
import { useProjects } from '../hooks/useProjects'
import ProjectCard from '../components/projects/ProjectCard'
import CreateProjectModal from '../components/projects/CreateProjectModal'
import styles from './Dashboard.module.css'

export default function Dashboard() {
  const { projects, loading, error, createProject } = useProjects()
  const [showModal, setShowModal] = useState(false)

  if (loading) return <div className={styles.loading}>Cargando...</div>
  if (error) return <div className={styles.error}>{error}</div>

  return (
    <div className={styles.dashboard}>
      <div className={styles.header}>
        <h1>Mis Proyectos</h1>
        <button onClick={() => setShowModal(true)}>
          + Nuevo Proyecto
        </button>
      </div>

      {projects?.length === 0 ? (
        <div className={styles.empty}>
          <p>No tienes proyectos aún</p>
          <button onClick={() => setShowModal(true)}>
            Crear tu primer proyecto
          </button>
        </div>
      ) : (
        <div className={styles.grid}>
          {projects?.map(project => (
            <ProjectCard key={project.id} project={project} />
          ))}
        </div>
      )}

      {showModal && (
        <CreateProjectModal
          onClose={() => setShowModal(false)}
          onSubmit={async (data) => {
            await createProject(data)
            setShowModal(false)
          }}
        />
      )}
    </div>
  )
}

Hook useProjects

// src/hooks/useProjects.js
import { useState, useEffect, useCallback } from 'react'
import { projectService } from '../services/projectService'

export function useProjects() {
  const [projects, setProjects] = useState([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  async function loadProjects() {
    try {
      setLoading(true)
      const data = await projectService.getAll()
      setProjects(data)
      setError(null)
    } catch (err) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    loadProjects()
  }, [])

  const createProject = useCallback(async (projectData) => {
    const newProject = await projectService.create(projectData)
    setProjects(prev => [...prev, newProject])
    return newProject
  }, [])

  const updateProject = useCallback(async (id, updates) => {
    const updated = await projectService.update(id, updates)
    setProjects(prev => prev.map(p => p.id === id ? updated : p))
    return updated
  }, [])

  const deleteProject = useCallback(async (id) => {
    await projectService.delete(id)
    setProjects(prev => prev.filter(p => p.id !== id))
  }, [])

  return {
    projects,
    loading,
    error,
    createProject,
    updateProject,
    deleteProject,
    reload: loadProjects
  }
}

Servicio de proyectos

// src/services/projectService.js
import { api } from './api'

export const projectService = {
  getAll: () => api.get('/projects'),
  
  getById: (id) => api.get(`/projects/${id}`),
  
  create: (data) => api.post('/projects', data),
  
  update: (id, data) => api.put(`/projects/${id}`, data),
  
  delete: (id) => api.delete(`/projects/${id}`)
}

Modal de crear proyecto

// src/components/projects/CreateProjectModal.jsx
import { useState } from 'react'
import styles from './CreateProjectModal.module.css'

export default function CreateProjectModal({ onClose, onSubmit }) {
  const [name, setName] = useState('')
  const [description, setDescription] = useState('')
  const [loading, setLoading] = useState(false)

  async function handleSubmit(e) {
    e.preventDefault()
    setLoading(true)
    
    try {
      await onSubmit({ name, description })
    } finally {
      setLoading(false)
    }
  }

  return (
    <div className={styles.overlay} onClick={onClose}>
      <div className={styles.modal} onClick={e => e.stopPropagation()}>
        <h2>Nuevo Proyecto</h2>
        <form onSubmit={handleSubmit}>
          <div className={styles.field}>
            <label>Nombre</label>
            <input
              type="text"
              value={name}
              onChange={(e) => setName(e.target.value)}
              required
            />
          </div>

          <div className={styles.field}>
            <label>Descripción</label>
            <textarea
              value={description}
              onChange={(e) => setDescription(e.target.value)}
              rows={3}
            />
          </div>

          <div className={styles.actions}>
            <button type="button" onClick={onClose}>
              Cancelar
            </button>
            <button type="submit" disabled={loading}>
              {loading ? 'Creando...' : 'Crear'}
            </button>
          </div>
        </form>
      </div>
    </div>
  )
}

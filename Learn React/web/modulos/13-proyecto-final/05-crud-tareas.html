CRUD de Tareas, Etiquetas y Prioridades

Módulo 13 · Proyecto Final

Introducción

Las tareas son el núcleo de TaskFlow. Implementaremos un tablero tipo Kanban con columnas de estado, tarjetas arrastrables visualmente, etiquetas para categorizar y prioridades para destacar lo urgente.

Servicio de tareas

// src/services/taskService.js
import { api } from './api'

export const taskService = {
  getByProject: (projectId) =>
    api.get(`/projects/${projectId}/tasks`),

  getById: (id) =>
    api.get(`/tasks/${id}`),

  create: (projectId, data) =>
    api.post(`/projects/${projectId}/tasks`, data),

  update: (id, data) =>
    api.put(`/tasks/${id}`, data),

  delete: (id) =>
    api.delete(`/tasks/${id}`),

  move: (id, status) =>
    api.put(`/tasks/${id}/move`, { status })
}

Hook useTasks

// src/hooks/useTasks.js
import { useState, useEffect, useCallback } from 'react'
import { taskService } from '../services/taskService'

export function useTasks(projectId) {
  const [tasks, setTasks] = useState([])
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(null)

  async function loadTasks() {
    try {
      setLoading(true)
      const data = await taskService.getByProject(projectId)
      setTasks(data)
      setError(null)
    } catch (err) {
      setError(err.message)
    } finally {
      setLoading(false)
    }
  }

  useEffect(() => {
    loadTasks()
  }, [projectId])

  const createTask = useCallback(async (taskData) => {
    const newTask = await taskService.create(projectId, taskData)
    setTasks((prev) => [...prev, newTask])
    return newTask
  }, [projectId])

  const updateTask = useCallback(async (id, updates) => {
    const updated = await taskService.update(id, updates)
    setTasks((prev) => prev.map((t) => (t.id === id ? updated : t)))
    return updated
  }, [])

  const moveTask = useCallback(async (id, status) => {
    const updated = await taskService.move(id, status)
    setTasks((prev) => prev.map((t) => (t.id === id ? updated : t)))
    return updated
  }, [])

  const deleteTask = useCallback(async (id) => {
    await taskService.delete(id)
    setTasks((prev) => prev.filter((t) => t.id !== id))
  }, [])

  return {
    tasks,
    setTasks,
    loading,
    error,
    createTask,
    updateTask,
    moveTask,
    deleteTask,
    reload: loadTasks
  }
}

Tablero Kanban

// src/components/tasks/TaskBoard.jsx
import { TASK_STATUS } from '../../types/task'
import { useTasks } from '../../hooks/useTasks'
import TaskColumn from './TaskColumn'
import styles from './TaskBoard.module.css'

const COLUMNS = [
  { key: TASK_STATUS.TODO, title: 'Por hacer' },
  { key: TASK_STATUS.IN_PROGRESS, title: 'En progreso' },
  { key: TASK_STATUS.DONE, title: 'Completadas' }
]

export default function TaskBoard({ projectId }) {
  const { tasks, loading, moveTask, createTask } = useTasks(projectId)

  if (loading) return <div>Cargando tareas...</div>

  return (
    <div className={styles.board}>
      {COLUMNS.map((column) => (
        <TaskColumn
          key={column.key}
          title={column.title}
          status={column.key}
          tasks={tasks.filter((t) => t.status === column.key)}
          onMove={moveTask}
          onCreate={createTask}
        />
      ))}
    </div>
  )
}

Columna de tareas

// src/components/tasks/TaskColumn.jsx
import { useState } from 'react'
import TaskCard from './TaskCard'
import CreateTaskModal from './CreateTaskModal'
import styles from './TaskColumn.module.css'

export default function TaskColumn({ title, status, tasks, onMove, onCreate }) {
  const [showModal, setShowModal] = useState(false)

  return (
    <div className={styles.column}>
      <div className={styles.header}>
        <h3>{title}</h3>
        <span className={styles.count}>{tasks.length}</span>
      </div>

      <div className={styles.list}>
        {tasks.map((task) => (
          <TaskCard key={task.id} task={task} onMove={onMove} />
        ))}
      </div>

      <button
        className={styles.addButton}
        onClick={() => setShowModal(true)}
      >
        + Añadir tarea
      </button>

      {showModal && (
        <CreateTaskModal
          status={status}
          onClose={() => setShowModal(false)}
          onSubmit={async (data) => {
            await onCreate({ ...data, status })
            setShowModal(false)
          }}
        />
      )}
    </div>
  )
}

Tarjeta de tarea

// src/components/tasks/TaskCard.jsx
import { TASK_STATUS } from '../../types/task'
import styles from './TaskCard.module.css'

const PRIORITY_STYLES = {
  high: styles.priorityHigh,
  medium: styles.priorityMedium,
  low: styles.priorityLow
}

export default function TaskCard({ task, onMove }) {
  return (
    <article className={styles.card}>
      <div className={styles.header}>
        <span className={`${styles.priority} ${PRIORITY_STYLES[task.priority]}`}>
          {task.priority}
        </span>
      </div>

      <h4>{task.title}</h4>
      <p>{task.description}</p>

      {task.labels?.length > 0 && (
        <div className={styles.labels}>
          {task.labels.map((label) => (
            <span key={label} className={styles.label}>
              {label}
            </span>
          ))}
        </div>
      )}

      <div className={styles.actions}>
        {task.status !== TASK_STATUS.TODO && (
          <button onClick={() => onMove(task.id, TASK_STATUS.TODO)}>
            ←
          </button>
        )}
        {task.status !== TASK_STATUS.IN_PROGRESS && (
          <button onClick={() => onMove(task.id, TASK_STATUS.IN_PROGRESS)}>
            →
          </button>
        )}
        {task.status !== TASK_STATUS.DONE && (
          <button onClick={() => onMove(task.id, TASK_STATUS.DONE)}>
            ✓
          </button>
        )}
      </div>
    </article>
  )
}

Modal de crear tarea

// src/components/tasks/CreateTaskModal.jsx
import { useState } from 'react'
import styles from './CreateTaskModal.module.css'

export default function CreateTaskModal({ status, onClose, onSubmit }) {
  const [title, setTitle] = useState('')
  const [description, setDescription] = useState('')
  const [priority, setPriority] = useState('medium')
  const [labels, setLabels] = useState('')
  const [dueDate, setDueDate] = useState('')

  async function handleSubmit(e) {
    e.preventDefault()
    await onSubmit({
      title,
      description,
      priority,
      dueDate,
      labels: labels.split(',').map((l) => l.trim()).filter(Boolean)
    })
  }

  return (
    <div className={styles.overlay} onClick={onClose}>
      <div className={styles.modal} onClick={(e) => e.stopPropagation()}>
        <h2>Nueva Tarea</h2>
        <form onSubmit={handleSubmit}>
          <label>
            Título
            <input value={title} onChange={(e) => setTitle(e.target.value)} required />
          </label>

          <label>
            Descripción
            <textarea value={description} onChange={(e) => setDescription(e.target.value)} />
          </label>

          <label>
            Prioridad
            <select value={priority} onChange={(e) => setPriority(e.target.value)}>
              <option value="low">Baja</option>
              <option value="medium">Media</option>
              <option value="high">Alta</option>
            </select>
          </label>

          <label>
            Etiquetas (separadas por coma)
            <input value={labels} onChange={(e) => setLabels(e.target.value)} />
          </label>

          <label>
            Fecha límite
            <input type="date" value={dueDate} onChange={(e) => setDueDate(e.target.value)} />
          </label>

          <div className={styles.actions}>
            <button type="button" onClick={onClose}>Cancelar</button>
            <button type="submit">Crear</button>
          </div>
        </form>
      </div>
    </div>
  )
}

Resumen

  • Las tareas tienen estado, prioridad y etiquetas
  • useTasks centraliza el CRUD de tareas de un proyecto
  • TaskBoard muestra tres columnas: por hacer, en progreso y completadas
  • TaskCard permite mover tareas entre columnas
  • CreateTaskModal recoge todos los campos necesarios
  • Las etiquetas se guardan como array de strings

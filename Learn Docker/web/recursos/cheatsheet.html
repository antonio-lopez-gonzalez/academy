Cheatsheet Docker / kubectl / Helm

Referencia rápida 🐳 Docker · ☸ Kubernetes

Docker · contenedores e imágenes

# Imágenes
docker pull nginx:1.27-alpine      # descargar imagen
docker images                      # listar imágenes locales
docker rmi IMAGEN                  # eliminar imagen
docker tag app v1 app:1.2.0        # etiquetar

# Contenedores
docker run -d --name web -p 8080:80 nginx:1.27-alpine  # crear y ejecutar en background
docker run -it --rm alpine sh      # interactivo, elimina al salir (--rm)
docker ps                          # contenedores en marcha
docker ps -a                       # todos (incluidos parados)
docker stop web && docker rm web   # parar y eliminar
docker logs -f web                 # seguir los logs
docker exec -it web sh             # terminal dentro del contenedor
docker stats                       # consumo de CPU/memoria en directo
docker inspect web                 # metadatos completos (JSON)

# Puertos, red y datos
docker run -p 8080:80 -e MODE=dev --network mynet -v vol1:/data app
docker network create mynet        # crear red bridge personalizada
docker volume create vol1          # crear volumen

Docker · Dockerfile y Compose

# Dockerfile multi-stage: compila en una etapa, publica en otra
FROM node:20-alpine AS build
WORKDIR /app
COPY package*.json ./
RUN npm ci --omit=dev
COPY . .
RUN npm run build

FROM nginx:1.27-alpine
COPY --from=build /app/dist /usr/share/nginx/html
EXPOSE 80
HEALTHCHECK CMD wget -qO- http://localhost/ || exit 1
# Compose
docker compose up -d               # levantar el stack
docker compose up -d --build       # reconstruir imágenes
docker compose down                # apagar (sin -v conserva volúmenes)
docker compose down -v             # apagar y eliminar volúmenes
docker compose ps                  # estado de servicios
docker compose logs -f api         # logs de un servicio
docker compose config              # validar y expandir el compose.yaml
docker compose exec api sh         # terminal en un servicio

kind · clúster local

kind create cluster --name curso   # crear clúster (Kubernetes en Docker)
kind get clusters                  # listar clústeres
kind delete cluster --name curso   # eliminar clúster
kind create cluster --config kind-config.yaml  # clúster con configuración

# Cargar una imagen local en el clúster (evita salir a un registry)
kind load docker-image api:1.0.0

kubectl · operación del clúster

kubectl config get-contexts        # ver contextos (clúster/nodo)
kubectl get nodes                  # nodos del clúster
kubectl get pods -A                # pods en todos los namespaces
kubectl get all -n harbortasks     # todos los recursos de un namespace

kubectl apply -f deploy.yaml       # aplicar manifiesto (declarativo+idempotente)
kubectl delete -f deploy.yaml      # eliminar lo declarado
kubectl get deployment api         # ver deployment
kubectl describe pod api-xxxxx     # eventos y detalles de un pod
kubectl logs -f deployment/api     # logs de un pod de un deployment
kubectl exec -it pod -- sh         # terminal dentro de un pod
kubectl port-forward svc/api 3000:3000  # túnel local a un service

kubectl rollout status deployment/api    # estado del despliegue
kubectl rollout undo deployment/api      # deshacer el último rollout
kubectl scale deployment api --replicas=3

Helm · charts

helm create misvalores            # esqueleto de chart
helm lint ./mi-chart              # validar el chart
helm template ./mi-chart          # renderizar YAML sin instalar (dry-run)
helm install release ./mi-chart   # instalar una release
helm install release ./mi-chart --values prod.yaml  # con valores propios
helm upgrade release ./mi-chart   # actualizar
helm rollback release 2           # volver a la revisión 2
helm list                         # releases instaladas
helm uninstall release            # desinstalar

Troubleshooting rápido

docker ps -a                       # ¿está el contenedor? ¿Exit code?
docker logs CONTAINER              # ¿qué imprimió al fallar?
docker inspect CONTAINER           # ¿restart policy, health, puertos?

kubectl get pods -o wide           # ¿dónde corre y en qué estado?
kubectl describe pod POD           # events: ImagePullBackOff, CrashLoopBackOff…
kubectl logs POD --previous        # logs de la ejecución anterior (crash)
kubectl get events --sort-by=.lastTimestamp

# Estados típicos
# Pending       → recursos/almacenamiento/PVC no disponible
# ImagePullBackOff → imagen no existe o sin acceso al registry
# CrashLoopBackOff → el proceso sale con error (mirar logs)
# OOMKilled     → se quedó sin memoria (limits)

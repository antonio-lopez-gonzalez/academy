Cheatsheet de TypeScript

Recurso Referencia rapida de TS 7 (compatible 5.9+)
Como usar esta chuleta

Es la referencia de bolsillo del curso: copia la sintaxis y ajusta los nombres. Todo el codigo asume strict activado. Para el "por que" de cada patron, abre la leccion correspondiente del sidebar.

1 · Tipos basicos

primitivos, arrays y tuplas
let nombre: string = "Ada";            // strings
let edad: number = 36;                 // numeros (int/float/doble)
let activo: boolean = true;            // booleanos
let n: bigint = 10n;                   // enteros grandes
let nada: undefined = undefined;
let vacio: null = null;

let ids: number[] = [1, 2, 3];         // array de numeros
let palabras: Array<string> = ["a"];   // forma alternativa
let par: [string, number] = ["x", 1];  // TUPLA: posiciones fijas
let fijo: readonly [string, number] = ["x", 1]; // tupla inmutable

const fila: readonly string[] = ["a"]; // array readonly

2 · Objetos e interfaces

interfaces, opcionales, index signatures
interface Usuario {
  readonly id: string;      // solo lectura (a nivel de tipos)
  nombre: string;
  email?: string;           // opcional
}

type Punto = { x: number; y: number };   // alias de objeto

type CualquierDato = { [clave: string]: number }; // index signature

// Intersection: suma de contratos
type Escribible = Usuario & { ultimoAcceso: Date };

// Union discriminada: el discriminante "tipo"
type Evento =
  | { tipo: "clic"; x: number; y: number }
  | { tipo: "tecla"; codigo: string };

3 · Unions, narrowing y literal types

reducir tipos con guardas
type Resultado = string | number;

function formatear(v: Resultado): string {
  if (typeof v === "string") return v.toUpperCase(); // narrowing por typeof
  return v.toFixed(2);                               // aqui v: number
}

// Type guard propio (predicado de tipo)
function esUsuario(x: unknown): x is Usuario {
  return typeof x === "object" && x !== null && "id" in x;
}

// Exhaustividad con never
function assertNever(v: never): never { throw new Error("caso inesperado"); }
as const y satisfies
const estados = ["pendiente", "hecho"] as const; // readonly literal[]

type Estado = (typeof estados)[number];           // "pendiente" | "hecho"

const config = {
  modo: "prod",
  retries: 3,
} satisfies Record<string, string | number>;      // valida SIN perder literales

4 · Funciones

parametros, callbacks y this
function sumar(a: number, b = 10): number {
  return a + b;
}

const restar: (a: number, b: number) => number = (a, b) => a - b;

type Callback = (err: Error | null, dato?: string) => void;

function conThis(this: { valor: number }, delta: number): number {
  return this.valor + delta;
}

// Rest + opcional
function juntar(separador = ", ", ...partes: string[]): string {
  return partes.join(separador);
}

5 · Genericos

parametros de tipo con restriccion
function primero<T>(arr: T[]): T | undefined {
  return arr[0];
}

function identidad<T extends { id: string }>(x: T): T {
  return x;
}

async function cargar<T>(url: string): Promise<T> {
  const res = await fetch(url);
  if (!res.ok) throw new Error("HTTP " + res.status);
  return (await res.json()) as T;
}
keyof, indexed y mapped
type Claves = keyof Usuario;              // "id" | "nombre" | "email"
type Nombre = Usuario["nombre"];          // string (indexed access)

type Opcional<T> = { [K in keyof T]?: T[K] };     // mapped: todo opcional
type SoloLectura<T> = { readonly [K in keyof T]: T[K] };
conditional types e infer
type SinNull<T> = T extends null | undefined ? never : T;

type TipoDePromesa<T> = T extends Promise<infer U> ? U : T; // extrae el tipo

6 · Utility types (los mas usados)

transformar tipos ya definidos
type Base = { id: string; titulo: string; ok: boolean };

type Parcial = Partial<Base>;            // todo opcional
type Obligatorio = Required<Parcial>;    // todo obligatorio
type ConId = Pick<Base, "id" | "titulo">;          // solo esos campos
type SinOk = Omit<Base, "ok">;                   // todos menos ok
type Registro = Record<string, number>;           // { [k]: number }
type Excluido = Exclude<"a" | "b" | "c", "a">;    // "b" | "c"
type Extraido = Extract<"a" | "b", "b" | "c">;    // "b"

function f(n: number, s: string): Promise<boolean> { return Promise.resolve(true); }
type Retorno = ReturnType<typeof f>;      // Promise<boolean>
type Args = Parameters<typeof f>;         // [number, string]
type Resuelto = Awaited<Retorno>;         // boolean
type SinNulo = NonNullable<string | null>; // string

7 · any, unknown, never

cuando usar cada uno
const peligroso: any = JSON.parse("{}");      // NO: escapa el chequeo
const dudoso: unknown = JSON.parse("{}");     // SI: hay que narrowing antes

if (typeof dudoso === "object" && dudoso !== null) {
  // aqui dudoso es "object no nulo"
}

function error(): never {                     // nunca retorna
  throw new Error("boom");
}

let vacio: never[] = [];                      // array vacio: never[]

8 · Template literals y branded types

strings tipados y marcas
type Ruta = `/tarea/${string}`;
const r1: Ruta = "/tarea/abc";      // ok
const r2: Ruta = "/otra";           // error

type Estado = "pendiente" | "hecho";
type Aviso = `tarea:${Estado}`;     // "tarea:pendiente" | "tarea:hecho"

// Branded type: separa strings con la misma forma
type IdUsuario = string & { readonly marca: unique symbol };
declare function crearId(s: string): IdUsuario;

9 · Modulos y configuracion

import/export con tipos
import type { Usuario } from "./modelo";   // solo tipos (borrado en build)
export type { Usuario };
import { crearUsuario } from "./api";        // valores (codigo real)

10 · TypeScript con React

props, eventos y hooks tipados
interface BotonProps {
  etiqueta: string;
  onClic: (id: number) => void;
}

export function Boton({ etiqueta, onClic }: BotonProps) {
  return <button onClick={(e) => { e.preventDefault(); onClic(1); }}>
    {etiqueta}
  </button>;
}

// Hooks tipados con genericos
const [items, setItems] = useState<string[]>([]);
useEffect(() => {
  let vivo = true;
  // ... fetch
  return () => { vivo = false; };
}, []);

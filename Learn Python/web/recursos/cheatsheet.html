Cheatsheet de Python

Referencia rápida 🐍 Python 3.12+

Tipos y variables

# Asignación: un nombre apunta a un objeto (no es una "caja")
nombre = "Ada"            # str
edad = 36                 # int
altura = 1.72             # float
activo = True             # bool
nada = None               # NoneType (ausencia intencional)

type(edad)                # <class 'int'>
isinstance(edad, int)     # True (preferido para comprobaciones)

# is = identidad; == = igualdad de valor
a = [1, 2]; b = [1, 2]
a == b                    # True  (mismo contenido)
a is b                    # False (objetos distintos)

# Solo usa `is` con None y centinelas
valor is None             # True

# Anotaciones (no se comprueban en runtime; para mypy)
def doblar(n: int) -> int:
    return n * 2

Strings y f-strings

s = "hola mundo"
s.upper()                 # 'HOLA MUNDO'
s.split()                 # ['hola', 'mundo']
"-".join(["a", "b"])      # 'a-b'
s[0], s[-1]               # ('h', 'o')
s[0:4]                    # 'hola' (slicing)

nombre, edad = "Ada", 36
f"{nombre} tiene {edad} años"          # 'Ada tiene 36 años'
f"{edad:04d}"                          # '0036' (relleno con ceros)
f"{3.14159:.2f}"                       # '3.14'
f"{nombre=}"                           # "nombre='Ada'" (debug)

"  hola  ".strip()        # 'hola'
"abc".startswith("ab")    # True

Colecciones

# Lista (ordenada, mutable)
frutas = ["pera", "manzana"]
frutas.append("uva")
frutas[0] = "kiwi"

# Tupla (ordenada, inmutable)
punto = (3, 4)
x, y = punto              # desempaquetado

# Diccionario (clave-valor, mutable)
stock = {"pera": 3, "uva": 0}
stock["kiwi"] = 5
stock.get("mango", 0)     # 0 si no existe (evita KeyError)
for fruta, cantidad in stock.items():
    ...

# Conjunto (sin duplicados, sin orden)
letras = {"a", "b", "a"}   # {'a', 'b'}
letras.add("c")
{"a", "b"} & {"b", "c"}   # intersección → {'b'}

Funciones

def saludar(nombre="mundo"):       # default inmutable, siempre
    return f"Hola, {nombre}"

def suma(*numeros):                # *args → tupla
    return sum(numeros)

def config(**opciones):            # **kwargs → dict
    return opciones

# NUNCA un mutable por defecto (bug clásico)
def malo(items=[]):                # ❌ se comparte entre llamadas
    items.append(1)
    return items

def bien(items=None):              # ✅
    items = items or []
    items.append(1)
    return items

cuadrado = lambda x: x ** 2        # lambda: solo para lo trivial

Comprehensions y generadores

dobles = [n * 2 for n in range(5)]          # [0, 2, 4, 6, 8]
pares = [n for n in range(10) if n % 2 == 0]
cuadrados = {n: n ** 2 for n in range(3)}   # {0:0, 1:1, 2:4}
unicos = {c.lower() for c in "Hola"}

# Generador: perezoso, no ocupa memoria
def contador(n):
    for i in range(n):
        yield i

sum(contador(1_000_000))           # no crea una lista gigante

# Slicing y copias
copia = frutas[:]                  # copia superficial
import copy
profunda = copy.deepcopy(objeto)

Clases y dataclasses

class Item:
    def __init__(self, title: str, borrowed: bool = False):
        self.title = title
        self.borrowed = borrowed

    def __repr__(self) -> str:      # para depurar
        return f"Item({self.title!r}, borrowed={self.borrowed})"

# Forma moderna y concisa
from dataclasses import dataclass, field

@dataclass
class Libro:
    title: str
    year: int | None = None
    tags: list[str] = field(default_factory=list)   # NUNCA lista compartida

@dataclass(frozen=True)            # inmutable y hashable
class Punto:
    x: int
    y: int

Errores, ficheros y JSON

try:
    valor = int(entrada)
except ValueError as err:          # captura específica, no `except:`
    print(f"no es un número: {err}")
else:
    print("ok")
finally:
    print("siempre")

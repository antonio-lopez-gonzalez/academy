Variables y tipos

// Declaración
const PI = 3.14;        // constante: no se puede reasignar (usar por defecto)
let contador = 0;       // variable reasignable, scope de bloque
var viejo = 1;          // 🚫 evitar: scope de función, hoisting confuso

// Tipos primitivos
const texto = 'hola';       // string
const numero = 42;          // number (todo son números, no hay int/float)
const big = 123n;           // bigint (enteros enormes)
const bool = true;          // boolean
const nulo = null;          // null (ausencia intencional)
const indef = undefined;    // undefined (sin asignar)
const simbolo = Symbol('x');// symbol (claves únicas)

// typeof: comprobar tipo (cuidado: typeof null === 'object' es un bug histórico)
console.log(typeof numero); // 'number'

Funciones

// Declarada (hoisted: se puede usar antes de su definición)
function sumar(a, b) { return a + b; }

// Arrow function (no tiene su propio this, no se puede usar como método de objeto)
const restar = (a, b) => a - b;
const cuadrado = (x) => x * x;    // un parámetro: se pueden omitir paréntesis

// Parámetros por defecto
function saludar(nombre = 'mundo') { return `Hola ${nombre}`; }

// Rest (...recoge el resto en un array) y spread (...expande)
function total(primer, ...resto) { return primer + resto.reduce((a, b) => a + b, 0); }
const nums = [1, 2, 3];
console.log(Math.max(...nums)); // 3

// Funciones de orden superior: reciben o devuelven funciones
const doble = nums.map((n) => n * 2);

Arrays

const frutas = ['manzana', 'pera', 'uva'];

// Añadir / quitar
frutas.push('naranja');          // al final
frutas.pop();                    // quita el último
frutas.unshift('kiwi');          // al principio
frutas.shift();                  // quita el primero
const sub = frutas.slice(0, 2);  // copia (no muta)
frutas.splice(1, 1, 'plátano');  // reemplaza (muta)

// Iteración y transformación (no mutan, crean nuevo array)
frutas.forEach((f) => console.log(f));
const mayusculas = frutas.map((f) => f.toUpperCase());
const conA = frutas.filter((f) => f.includes('a'));
const primera = frutas.find((f) => f.startsWith('p'));
const hayUva = frutas.some((f) => f === 'uva');
const todasCortas = frutas.every((f) => f.length < 6);
const total = [1, 2, 3].reduce((acc, n) => acc + n, 0); // 6
frutas.sort();                   // ordena (muta)
const copia = [...frutas];       // clonar (shallow)
const [primero, segundo] = frutas; // desestructuración

// Iterator helpers (ES2025): map/filter/take/drop sobre CUALQUIER iterador
Iterator.from([1, 2, 3, 4, 5])
  .filter(n => n % 2 !== 0)     // 1, 3, 5
  .map(n => n * 10)             // 10, 30, 50
  .take(2)                      // 10, 30
  .toArray();                   // [10, 30] — evaluación perezosa, sin arrays temporales

// Operaciones de conjunto en Set (ES2025)
new Set(['a', 'b']).union(new Set(['b', 'c']));         // Set {'a','b','c'}
new Set(['a', 'b']).intersection(new Set(['b', 'c']));  // Set {'b'}
new Set(['a', 'b']).isSubsetOf(new Set(['a', 'b', 'c'])); // true

Objetos y clases

// Literal
const usuario = { nombre: 'Ana', edad: 30 };
usuario.ciudad = 'Madrid';           // añadir propiedad
const { nombre, edad } = usuario;    // desestructurar
const clon = { ...usuario };         // clonar (shallow)
const resumen = { ...usuario, rol: 'admin' }; // spread + nueva propiedad

// Optional chaining y nullish coalescing (ES2020)
usuario?.perfil?.avatar;             // undefined en vez de TypeError
const valor = usuario.nombre ?? 'anonimo'; // usa 'anonimo' solo si es null/undefined

// Clase (azúcar sobre prototipos)
class Coche {
  #vin;                             // propiedad privada (ES2022)
  constructor(marca) { this.marca = marca; this.#vin = '123'; }
  arrancar() { return `${this.marca} arrancado`; }
  static crear(marca) { return new Coche(marca); } // método estático
}
class Electrico extends Coche {
  constructor(marca) { super(marca); }   // llama al constructor padre
  cargar() { return 'cargando…'; }
}

Asincronía

// Promesas
fetch('/api/datos')
  .then((res) => { if (!res.ok) throw new Error('HTTP ' + res.status); return res.json(); })
  .then((datos) => console.log(datos))
  .catch((err) => console.error(err))
  .finally(() => console.log('terminado'));

// async/await (la forma moderna y más legible)
async function cargarDatos() {
  try {
    const res = await fetch('/api/datos');
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    return await res.json();
  } catch (err) {
    console.error(err);
    throw err; // relanzar si el llamador lo necesita
  }
}

// Varias promesas
const [a, b] = await Promise.all([p1, p2]);   // todas (falla si una falla)
const ganadora = await Promise.race([p1, p2]); // la primera en resolver
Promise.allSettled([p1, p2]);                 // todas, sin fallar

// Promise.try (ES2025): envuelve funciones síncronas/async/que lanzan en una promesa
Promise.try(miFuncion).then((v) => console.log(v)).catch((e) => console.error(e));

// RegExp.escape (ES2025): escapa texto para usarlo en un patrón regex
new RegExp(RegExp.escape('precio (con) $paréntesis$'), 'g');

// Cancelar peticiones
const controller = new AbortController();
fetch('/api/datos', { signal: controller.signal });
controller.abort(); // cancela
